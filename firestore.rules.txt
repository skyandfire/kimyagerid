rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Herkes okur, yalnızca yönetici yazar. BURAYA_UID yerine kendi Kullanıcı UID'ini yapıştır.
    function admin() { return request.auth != null && request.auth.uid == "BURAYA_UID"; }
    match /chemicals/{cas} { allow read: if true; allow write: if admin(); }
    match /settings/{doc}  { allow read: if true; allow write: if admin(); }
  }
}
